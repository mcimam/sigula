/* ---------------- helpers ---------------- */
function fmtDate(d) {
  if (!d) return "-";
  const dt = new Date(d + "T00:00:00");
  return dt.toLocaleDateString("id-ID", { day: "2-digit", month: "short", year: "numeric" });
}
function daysBetween(a, b) {
  const A = new Date(a + "T00:00:00");
  const B = new Date(b + "T00:00:00");
  return Math.round((B - A) / 86400000);
}
function addDays(dateStr, n) {
  const d = new Date(dateStr + "T00:00:00");
  d.setDate(d.getDate() + n);
  return d.toISOString().slice(0, 10);
}
function uid(prefix) {
  return prefix + "-" + Math.random().toString(36).slice(2, 9);
}
function classNames(...a) {
  return a.filter(Boolean).join(" ");
}

/* ---------------- small UI atoms ---------------- */
function Badge({ children, tone = "slate" }) {
  const tones = {
    slate: "bg-slate-100 text-slate-700 border-slate-300",
    teal: "bg-teal-100 text-teal-800 border-teal-300",
    amber: "bg-amber-100 text-amber-800 border-amber-300",
    red: "bg-rose-100 text-rose-700 border-rose-300",
    stone: "bg-stone-200 text-stone-600 border-stone-400",
    green: "bg-emerald-100 text-emerald-800 border-emerald-300",
  };
  return (
    <span className={classNames("inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-xs font-medium", tones[tone])}>
      {children}
    </span>
  );
}

function Card({ children, className = "" }) {
  return (
    <div className={classNames("rounded-xl border border-slate-200 bg-white shadow-sm", className)}>
      {children}
    </div>
  );
}

function StatTile({ label, value, sub, tone = "slate", icon: Icon }) {
  const toneMap = {
    slate: "text-slate-900",
    teal: "text-teal-700",
    amber: "text-amber-700",
    red: "text-rose-700",
  };
  return (
    <Card className="p-4 flex items-start justify-between">
      <div>
        <div className="text-xs uppercase tracking-wide text-slate-400 font-semibold">{label}</div>
        <div className={classNames("text-2xl font-bold tabular-nums mt-1", toneMap[tone])}>{value}</div>
        {sub && <div className="text-xs text-slate-400 mt-1">{sub}</div>}
      </div>
      {Icon && (
        <div className={classNames("h-9 w-9 rounded-lg flex items-center justify-center shrink-0",
          tone === "teal" ? "bg-teal-50" : tone === "amber" ? "bg-amber-50" : tone === "red" ? "bg-rose-50" : "bg-slate-50")}>
          <Icon className={classNames("h-5 w-5", toneMap[tone])} />
        </div>
      )}
    </Card>
  );
}

function Button({ children, onClick, variant = "primary", size = "md", className = "", disabled, type = "button" }) {
  const base = "inline-flex items-center justify-center gap-1.5 rounded-lg font-medium transition-colors disabled:opacity-40 disabled:cursor-not-allowed";
  const sizes = { sm: "text-xs px-2.5 py-1.5", md: "text-sm px-3.5 py-2" };
  const variants = {
    primary: "bg-teal-700 text-white hover:bg-teal-800",
    secondary: "bg-white text-slate-700 border border-slate-300 hover:bg-slate-50",
    ghost: "text-slate-600 hover:bg-slate-100",
    danger: "bg-rose-600 text-white hover:bg-rose-700",
    amber: "bg-amber-600 text-white hover:bg-amber-700",
  };
  return (
    <button type={type} onClick={onClick} disabled={disabled}
      className={classNames(base, sizes[size], variants[variant], className)}>
      {children}
    </button>
  );
}

    ]);
    let msg = `Notifikasi terkirim ke ${affectedSalesmanIds.length} salesman (${toNotify.length} customer) via WhatsApp` +
      (affectedSupervisorIds.length ? `, dan ringkasannya diteruskan ke ${affectedSupervisorIds.length} supervisor terkait.` : ".");
    if (salesmenMissingPhone.length > 0 || supervisorsMissingPhone.length > 0) {
      const parts = [];
      if (salesmenMissingPhone.length) parts.push(`${salesmenMissingPhone.length} salesman (${salesmenMissingPhone.map((s) => s.nama).join(", ")})`);
      if (supervisorsMissingPhone.length) parts.push(`${supervisorsMissingPhone.length} supervisor (${supervisorsMissingPhone.map((s) => s.nama).join(", ")})`);
      msg += ` ⚠️ Perhatian: nomor WA belum diisi untuk ${parts.join(" dan ")} — notifikasi ke mereka tidak akan benar-benar terkirim di produksi. Lengkapi di Master Salesman/Supervisor.`;
    }
    notify(msg);
  }

  function updateCustomer(id, patch) {
    setCustomers((prev) => prev.map((c) => (c.id === id ? { ...c, ...patch } : c)));
  }

  function assignSupervisor(salesmanId, supervisorId) {
    setSalesmen((prev) => prev.map((s) => (s.id === salesmanId ? { ...s, supervisor_id: supervisorId } : s)));
    notify("Supervisor untuk salesman berhasil diperbarui.");
  }

  function updateSalesmanPhone(salesmanId, phone) {
    setSalesmen((prev) => prev.map((s) => (s.id === salesmanId ? { ...s, nomor_wa: phone } : s)));
  }

  function updateSupervisorPhone(supervisorId, phone) {
    setSupervisors((prev) => prev.map((s) => (s.id === supervisorId ? { ...s, nomor_wa: phone } : s)));
  }

  function toggleSalesmanStatus(salesmanId) {
    setSalesmen((prev) =>
      prev.map((s) => (s.id === salesmanId ? { ...s, status: s.status === "Aktif" ? "Inactive" : "Aktif" } : s))
    );
  }

  function mutateCustomer(customerId, newSalesmanId) {
    const cust = customers.find((c) => c.id === customerId);
    if (!cust || cust.salesman_id === newSalesmanId) return;
    setMutationLog((prev) => [
      { id: uid("MUT"), customer_id: customerId, dari: cust.salesman_id, ke: newSalesmanId, tanggal: simDate, oleh: "Admin" },
      ...prev,
    ]);
    updateCustomer(customerId, { salesman_id: newSalesmanId });
    notify("Customer berhasil dimutasi ke salesman baru.");
  }

  function setCycleDays(customerId, days) {
    updateCustomer(customerId, { order_cycle_days: Math.max(1, Number(days) || 1) });
  }

  function submitReason(customerId, kode) {
    const cust = customers.find((c) => c.id === customerId);
    setReasonLog((prev) => [
      { id: uid("RSN"), customer_id: customerId, salesman_id: cust.salesman_id, tanggal: simDate, kode_alasan: kode },
      ...prev,
    ]);
    if (kode === 3) {
      updateCustomer(customerId, { status_customer: "Inactive", handled_on: simDate, notified: false });
      setStatusLog((prev) => [
        { id: uid("STL"), customer_id: customerId, tipe: "Inactive", tanggal: simDate, oleh: `Salesman (${cust.salesman_id})` },
        ...prev,
      ]);
      notify(`${cust.nama} ditandai Inactive (Sudah Bangkrut). Tidak akan dinotifikasi lagi.`);
    } else {
      updateCustomer(customerId, { handled_on: simDate, notified: false });
      notify(`Alasan "${REASON_LABELS[kode]}" tercatat untuk ${cust.nama}. Customer ini akan masuk lagi ke daftar kalau Admin trigger notifikasi berikutnya dan masih melewati siklus.`);
    }
  }

  function recordNewOrder(customerId) {
    const cust = customers.find((c) => c.id === customerId);
    const wasInactive = cust.status_customer === "Inactive";
    updateCustomer(customerId, { last_order_date: simDate, status_customer: "Aktif", handled_on: null, notified: false });
    setOrderHistoryLog((prev) => [
      { id: uid("ORD"), customer_id: customerId, tanggal_order: simDate, sumber: "Manual (dicatat Salesman)", tanggal_input: simDate },
      ...prev,
    ]);
    if (wasInactive) {
      setStatusLog((prev) => [
        { id: uid("STL"), customer_id: customerId, tipe: "Reaktivasi Otomatis (order baru)", tanggal: simDate, oleh: "Sistem" },
        ...prev,
      ]);
      notify(`${cust.nama} order lagi → status otomatis kembali Aktif.`);
    } else {
      notify(`Order baru tercatat untuk ${cust.nama}.`);
    }
  }

  function manualReactivate(customerId) {
    const cust = customers.find((c) => c.id === customerId);
    updateCustomer(customerId, { status_customer: "Aktif", handled_on: null, notified: false });
    setStatusLog((prev) => [
      { id: uid("STL"), customer_id: customerId, tipe: "Reaktivasi Manual", tanggal: simDate, oleh: "Supervisor" },
      ...prev,
    ]);
    notify(`${cust.nama} diaktifkan kembali secara manual oleh Supervisor.`);
  }

  /* -------- excel export -------- */
  function exportExcel(rows, filename, sheetName = "Data") {
    const ws = XLSX.utils.json_to_sheet(rows);
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, sheetName);
    XLSX.writeFile(wb, filename);
    notify(`File ${filename} sedang diunduh.`);
  }

  function exportSupervisorReport(supId) {
    const supName = supervisors.find((s) => s.id === supId)?.nama || "Supervisor";
    const teamIds = supervisorOf(supId).map((s) => s.id);
    const teamCustomers = enrichedCustomers.filter((c) => teamIds.includes(c.salesman_id));

    const summaryRows = supervisorOf(supId).map((s) => {
      const cs = enrichedCustomers.filter((c) => c.salesman_id === s.id);
      return {
        Salesman: s.nama,
        "Status Salesman": s.status,
        "Total Customer": cs.length,
        Aktif: cs.filter((c) => c.status_customer === "Aktif").length,
        Inactive: cs.filter((c) => c.status_customer === "Inactive").length,
        "Perlu Follow Up": cs.filter((c) => c.flagNotifikasi).length,
      };
    });
    const detailRows = teamCustomers.map((c) => ({
      Customer: c.nama,
      Salesman: c.salesmanNama,
      "Tipe Customer": c.tipe_customer,
      "Tgl Order Terakhir": c.last_order_date || "-",
      "Hari Belum Order": c.daysSince ?? "-",
      "Siklus Normal (hari)": c.order_cycle_days,
      Status: c.status_customer,
      "Perlu Follow Up": c.flagNotifikasi ? "Ya" : "Tidak",
    }));

    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(summaryRows), "Ringkasan Salesman");
    XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(detailRows), "Detail Customer");
    XLSX.writeFile(wb, `Laporan_${supName}_${simDate}.xlsx`);
    notify("Laporan Excel sedang diunduh.");
  }

  function exportManagementReport() {
    const perSupervisor = supervisors.map((sup) => {
      const teamIds = supervisorOf(sup.id).map((s) => s.id);
      const cs = enrichedCustomers.filter((c) => teamIds.includes(c.salesman_id));
      return {
        Supervisor: sup.nama,
        "Jml Salesman": supervisorOf(sup.id).length,
        "Total Customer": cs.length,
        Aktif: cs.filter((c) => c.status_customer === "Aktif").length,
        Inactive: cs.filter((c) => c.status_customer === "Inactive").length,
        "Perlu Follow Up": cs.filter((c) => c.flagNotifikasi).length,
      };
    });
    const reasonCounts = { 1: 0, 2: 0, 3: 0 };
    reasonLog.forEach((r) => (reasonCounts[r.kode_alasan] = (reasonCounts[r.kode_alasan] || 0) + 1));
    const reasonRows = Object.keys(REASON_LABELS).map((k) => ({
      Alasan: REASON_LABELS[k],
      "Jumlah Kejadian": reasonCounts[k] || 0,
    }));
    const salesmanRows = salesmen.map((s) => {
      const cs = enrichedCustomers.filter((c) => c.salesman_id === s.id);
      const flagged = cs.filter((c) => c.flagNotifikasi).length;
      const responded = reasonLog.filter((r) => r.salesman_id === s.id).length;
      return {
        Salesman: s.nama,
        Supervisor: supervisors.find((sp) => sp.id === s.supervisor_id)?.nama || "-",
        "Total Customer": cs.length,
        "Perlu Follow Up": flagged,
        "Alasan Terinput (all-time)": responded,
      };
    });

    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(perSupervisor), "Per Supervisor");
    XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(salesmanRows), "Per Salesman");
    XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(reasonRows), "Breakdown Alasan");
    XLSX.writeFile(wb, `Summary_Report_Management_${simDate}.xlsx`);
    notify("Summary report untuk management sedang diunduh.");
  }

  /* -------- excel import (migrasi, format sheet-per-salesman) -------- */
  function parseImportFile(e) {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (evt) => {
      try {
        const wb = XLSX.read(evt.target.result, { type: "array", cellDates: true });
        const unknownSheets = []; // sheet yang tidak cocok dengan salesman manapun di master data
        const newCustomers = []; // { sheetName, salesmanId (null jika sheet unknown), nama, tipe, lastOrder }
        const updatedCustomers = []; // { existingId, nama, salesmanNama, oldDate, newDate, wasInactive }
        let unchanged = 0;

        wb.SheetNames.forEach((sheetName) => {
          const ws = wb.Sheets[sheetName];
          const rows = XLSX.utils.sheet_to_json(ws, { header: 1, raw: false });
          let headerIdx = -1;
          for (let i = 0; i < Math.min(rows.length, 8); i++) {
            if (rows[i] && rows[i].some((cell) => String(cell).trim() === "Nama Konsumen")) {
              headerIdx = i;
              break;
            }
          }
          if (headerIdx === -1) return;
          const header = rows[headerIdx].map((h) => String(h || "").trim());
          const colName = header.indexOf("Nama Konsumen");
          const colStatus = header.indexOf("Status (Lama/Baru)");
          const colTgl = header.indexOf("Tgl Terakhir Order");

          const matchedSalesman = salesmen.find((s) => s.nama.toUpperCase() === sheetName.toUpperCase());
          if (!matchedSalesman && !unknownSheets.includes(sheetName)) {
            unknownSheets.push(sheetName);
          }

          for (let i = headerIdx + 1; i < rows.length; i++) {
            const row = rows[i];
            if (!row || !row[colName]) continue;
            const nama = String(row[colName]).trim();
            const statusRaw = colStatus >= 0 ? String(row[colStatus] || "") : "";
            const tipe = statusRaw.toLowerCase().includes("baru") ? "Baru" : "Lama";
            let lastOrder = null;
            if (colTgl >= 0 && row[colTgl]) {
              const parsed = new Date(row[colTgl]);
              if (!isNaN(parsed.getTime())) lastOrder = parsed.toISOString().slice(0, 10);
            }

            if (!matchedSalesman) {
              newCustomers.push({ sheetName, salesmanId: null, nama, tipe, lastOrder });
              continue;
            }

            const existing = customers.find(
              (c) => c.nama.toLowerCase() === nama.toLowerCase() && c.salesman_id === matchedSalesman.id
            );
            if (!existing) {
              newCustomers.push({ sheetName, salesmanId: matchedSalesman.id, nama, tipe, lastOrder });
            } else if (lastOrder && lastOrder !== existing.last_order_date) {
              updatedCustomers.push({
                existingId: existing.id,
                nama: existing.nama,
                salesmanNama: matchedSalesman.nama,
                oldDate: existing.last_order_date,
                newDate: lastOrder,
                tipe,
                wasInactive: existing.status_customer === "Inactive",
              });
            } else {
              unchanged++;
            }
          }
        });

        setPendingImport({
          fileName: file.name,
          unknownSheets,
          newCustomers,
          updatedCustomers,
          unchanged,
        });
      } catch (err) {
        notify("Gagal membaca file. Pastikan format sesuai template (sheet per salesman, kolom 'Nama Konsumen' dst).");
      }
    };
    reader.readAsArrayBuffer(file);
    e.target.value = "";
  }

  function cancelImport() {
    setPendingImport(null);
    notify("Import dibatalkan. Perbarui Master Data dulu jika perlu, lalu upload ulang.");
  }

  function confirmImport(treatUnknownSheetsAsNewSalesman) {
    if (!pendingImport) return;
    const { unknownSheets, newCustomers, updatedCustomers } = pendingImport;

    if (unknownSheets.length > 0 && !treatUnknownSheetsAsNewSalesman) {
      notify("Import dibatalkan karena masih ada sheet yang belum dikenali. Perbaiki dulu di Master Salesman, atau pilih 'Tambahkan sebagai salesman baru'.");
      setPendingImport(null);
      return;
    }

    const salesmenCopy = [...salesmen];
    const customersCopy = [...customers];
    const statusLogAdditions = [];
    const orderHistoryAdditions = [];
    let nextCustSeq = customers.length + 1;

    const sheetToSalesmanId = {};
    unknownSheets.forEach((sheetName) => {
      const newSm = { id: uid("SLM"), nama: sheetName.toUpperCase(), supervisor_id: null, nomor_wa: "", status: "Aktif" };
      salesmenCopy.push(newSm);
      sheetToSalesmanId[sheetName] = newSm.id;
    });

    newCustomers.forEach((nc) => {
      const salesmanId = nc.salesmanId || sheetToSalesmanId[nc.sheetName];
      const newId = `CUST-IMP-${nextCustSeq++}`;
      customersCopy.push({
        id: newId,
        nama: nc.nama,
        salesman_id: salesmanId,
        tipe_customer: nc.tipe,
        order_cycle_days: 30,
        status_customer: "Aktif",
        last_order_date: nc.lastOrder,
      });
      if (nc.lastOrder) {
        orderHistoryAdditions.push({
          id: uid("ORD"), customer_id: newId, tanggal_order: nc.lastOrder,
          sumber: "Import Excel (customer baru)", tanggal_input: simDate,
        });
      }
    });

    updatedCustomers.forEach((uc) => {
      const idx = customersCopy.findIndex((c) => c.id === uc.existingId);
      if (idx < 0) return;
      customersCopy[idx] = {
        ...customersCopy[idx],
        last_order_date: uc.newDate,
        tipe_customer: uc.tipe,
        status_customer: "Aktif",
        notified: false,
      };
      orderHistoryAdditions.push({
        id: uid("ORD"), customer_id: uc.existingId, tanggal_order: uc.newDate,
        sumber: "Import Excel", tanggal_input: simDate,
      });
      if (uc.wasInactive) {
        statusLogAdditions.push({
          id: uid("STL"), customer_id: uc.existingId,
          tipe: "Reaktivasi Otomatis (order baru dari import Excel)",
          tanggal: simDate, oleh: "Sistem (Import)",
        });
      }
    });

    setSalesmen(salesmenCopy);
    setCustomers(customersCopy);
    if (statusLogAdditions.length) setStatusLog((prev) => [...statusLogAdditions, ...prev]);
    if (orderHistoryAdditions.length) setOrderHistoryLog((prev) => [...orderHistoryAdditions, ...prev]);

    const reactivatedCount = updatedCustomers.filter((u) => u.wasInactive).length;
    notify(
      `Import diterapkan: ${newCustomers.length} customer baru, ${updatedCustomers.length} diperbarui tanggal ordernya` +
      (reactivatedCount ? ` (${reactivatedCount} otomatis kembali Aktif)` : "") +
      `, ${pendingImport.unchanged} tidak berubah` +
      (unknownSheets.length ? `, ${unknownSheets.length} salesman baru ditambahkan (${unknownSheets.join(", ")}) — atur supervisornya di tab Master Salesman.` : ".")
    );
    setPendingImport(null);
  }

  if (!loaded) {
    return (
      <div className="min-h-[500px] flex items-center justify-center text-slate-400 text-sm">
        Memuat data...
      </div>
    );
  }

  return (
    <div className="min-h-[700px] bg-slate-50 text-slate-800 font-sans">
      <TopBar
        role={role} setRole={setRole}
        simDate={simDate} advanceDay={advanceDay}
        resetAllData={resetAllData}
        activeSupervisorId={activeSupervisorId} setActiveSupervisorId={setActiveSupervisorId}
        activeSalesmanId={activeSalesmanId} setActiveSalesmanId={setActiveSalesmanId}
        salesmen={salesmen} supervisors={supervisors}
      />

      <div className="max-w-7xl mx-auto px-4 py-6">
        {role === "admin" && (
          <AdminView
            salesmen={salesmen} supervisors={supervisors} customers={enrichedCustomers}
            mutationLog={mutationLog} statusLog={statusLog} notificationLog={notificationLog} orderHistoryLog={orderHistoryLog}
            assignSupervisor={assignSupervisor}
            toggleSalesmanStatus={toggleSalesmanStatus}
            mutateCustomer={mutateCustomer}
            setCycleDays={setCycleDays}
            handleImportFile={parseImportFile}
            pendingImport={pendingImport}
            confirmImport={confirmImport}
            cancelImport={cancelImport}
            triggerNotifications={triggerNotifications}
            updateSalesmanPhone={updateSalesmanPhone}
            updateSupervisorPhone={updateSupervisorPhone}
          />
        )}
        {role === "supervisor" && (
          <SupervisorView
            supervisorId={activeSupervisorId}
            supervisors={supervisors}
            salesmen={supervisorOf(activeSupervisorId)}
            customers={enrichedCustomers}
            notificationLog={notificationLog}
            manualReactivate={manualReactivate}
            exportReport={() => exportSupervisorReport(activeSupervisorId)}
          />
        )}
        {role === "salesman" && (
          <SalesmanView
            salesman={salesmen.find((s) => s.id === activeSalesmanId)}
            customers={enrichedCustomers.filter((c) => c.salesman_id === activeSalesmanId)}
            simDate={simDate}
            submitReason={submitReason}
            recordNewOrder={recordNewOrder}
          />
        )}
        {role === "management" && (
          <ManagementView
            supervisors={supervisors}
            salesmen={salesmen}
            customers={enrichedCustomers}
            reasonLog={reasonLog}
            statusLog={statusLog}
            supervisorOf={supervisorOf}
            exportReport={exportManagementReport}
          />
        )}
      </div>

      {toast && (
        <div className="fixed bottom-5 left-1/2 -translate-x-1/2 bg-slate-900 text-white text-sm px-4 py-2.5 rounded-lg shadow-lg max-w-md text-center z-50">
          {toast}
        </div>
      )}
    </div>
  );
}

  const tabs = [

      {tab === "salesman" && <AdminSalesmanTab salesmen={salesmen} supervisors={supervisors} customers={customers} assignSupervisor={assignSupervisor} toggleSalesmanStatus={toggleSalesmanStatus} updateSalesmanPhone={updateSalesmanPhone} />}
      {tab === "supervisor" && <AdminSupervisorTab supervisors={supervisors} salesmen={salesmen} updateSupervisorPhone={updateSupervisorPhone} />}
      {tab === "customer" && <AdminCustomerTab salesmen={salesmen} customers={customers} mutateCustomer={mutateCustomer} setCycleDays={setCycleDays} />}
      {tab === "import" && <AdminImportTab handleImportFile={handleImportFile} pendingImport={pendingImport} confirmImport={confirmImport} cancelImport={cancelImport} />}
      {tab === "log" && <AdminLogTab mutationLog={mutationLog} statusLog={statusLog} orderHistoryLog={orderHistoryLog} customers={customers} salesmen={salesmen} />}
    </div>
  );
}

function AdminNotifyTab({ salesmen, customers, notificationLog, triggerNotifications }) {
  const eligible = customers.filter((c) => c.flagNotifikasi && !c.notified);
  const alreadyPending = customers.filter((c) => c.notified);

  const perSalesmanPreview = useMemo(() => {
    const map = {};
    eligible.forEach((c) => {
      map[c.salesman_id] = (map[c.salesman_id] || 0) + 1;
    });
    return salesmen
      .map((s) => ({ salesman: s, count: map[s.id] || 0 }))
      .filter((r) => r.count > 0)
      .sort((a, b) => b.count - a.count);
  }, [eligible, salesmen]);

  return (
    <div className="grid lg:grid-cols-3 gap-4">
      <Card className="p-5 lg:col-span-2">
        <div className="flex items-start justify-between gap-4 flex-wrap">
          <div>
            <div className="font-semibold text-slate-800 flex items-center gap-1.5">
              <Send className="h-4 w-4 text-teal-700" /> Trigger Notifikasi Manual
            </div>
            <p className="text-sm text-slate-500 mt-1 max-w-lg">
              Sistem <b>tidak</b> mengirim notifikasi otomatis. Notifikasi WhatsApp hanya terkirim
              ke salesman (dan ringkasannya diteruskan ke supervisor terkait) setelah Admin menekan
              tombol di bawah ini — kapan pun Admin mau.
            </p>
          </div>
        </div>

        <div className="grid grid-cols-2 gap-3 mt-4">
          <StatTile label="Siap Dinotifikasi" value={eligible.length} sub="belum pernah dikirim" tone="red" icon={AlertTriangle} />
          <StatTile label="Sudah Dikirim, Menunggu Tindak Lanjut" value={alreadyPending.length} tone="amber" icon={Clock} />
        </div>

        {perSalesmanPreview.length > 0 ? (
          <div className="mt-4 border border-slate-200 rounded-lg overflow-hidden">
            <table className="w-full text-sm">
              <thead className="bg-slate-50 text-slate-500 text-xs uppercase tracking-wide">
                <tr>
                  <th className="text-left px-3 py-2 font-semibold">Salesman</th>
                  <th className="text-left px-3 py-2 font-semibold">Nomor WA</th>
                  <th className="text-left px-3 py-2 font-semibold">Customer akan dinotif</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {perSalesmanPreview.map((r) => (
                  <tr key={r.salesman.id}>
                    <td className="px-3 py-2 font-medium text-slate-700">{r.salesman.nama}</td>
                    <td className="px-3 py-2">
                      {r.salesman.nomor_wa
                        ? <span className="text-slate-500 text-xs">{r.salesman.nomor_wa}</span>
                        : <Badge tone="red">Nomor belum diisi</Badge>}
                    </td>
                    <td className="px-3 py-2"><Badge tone="red">{r.count} customer</Badge></td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        ) : (
          <div className="mt-4 text-sm text-slate-400 border border-dashed border-slate-200 rounded-lg p-4 text-center">
            Tidak ada customer baru yang perlu dinotifikasi saat ini.
          </div>
        )}

        <Button
          className="mt-4"
          disabled={eligible.length === 0}
          onClick={triggerNotifications}
        >
          <Send className="h-4 w-4" />
          Kirim Notifikasi Sekarang{eligible.length > 0 ? ` (${perSalesmanPreview.length} salesman, ${eligible.length} customer)` : ""}
        </Button>
      </Card>

      <Card className="p-4">
        <div className="font-semibold text-sm text-slate-700 mb-3 flex items-center gap-1.5">
          <ListChecks className="h-4 w-4" /> Riwayat Pengiriman
        </div>
        {notificationLog.length === 0 ? (
          <div className="text-xs text-slate-400">Belum pernah ada notifikasi yang dikirim.</div>
        ) : (
          <div className="space-y-3 max-h-96 overflow-auto">
            {notificationLog.map((l) => (
              <div key={l.id} className="text-xs border-b border-slate-100 pb-2.5">
                <div className="font-medium text-slate-700">{fmtDate(l.tanggal)}</div>
                <div className="text-slate-400 mt-0.5">
                  {l.totalCustomers} customer · {l.salesmanCount} salesman · {l.supervisorIds.length} supervisor diberi ringkasan
                </div>
              </div>
            ))}
          </div>
        )}
      </Card>
    </div>
  );
}

function AdminCustomerTab({ salesmen, customers, mutateCustomer, setCycleDays }) {
  const [q, setQ] = useState("");
  const [page, setPage] = useState(0);
  const pageSize = 25;

  const filtered = useMemo(() => {
    if (!q.trim()) return customers;
    const s = q.toLowerCase();
    return customers.filter((c) => c.nama.toLowerCase().includes(s) || c.salesmanNama.toLowerCase().includes(s));
  }, [customers, q]);

  const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
  const pageRows = filtered.slice(page * pageSize, page * pageSize + pageSize);

  return (
    <div>
      <div className="flex items-center gap-2 mb-3">
        <div className="relative flex-1 max-w-sm">
          <Search className="h-3.5 w-3.5 text-slate-400 absolute left-2.5 top-1/2 -translate-y-1/2" />
          <input
            value={q}
            onChange={(e) => { setQ(e.target.value); setPage(0); }}
            placeholder="Cari nama customer atau salesman..."
            className="w-full text-sm border border-slate-300 rounded-lg pl-8 pr-3 py-2 focus:outline-none focus:ring-2 focus:ring-teal-600/30 focus:border-teal-600"
          />
        </div>
        <div className="text-xs text-slate-400">{filtered.length} customer</div>
      </div>

      <Card className="overflow-hidden">
        <table className="w-full text-sm">
          <thead className="bg-slate-50 text-slate-500 text-xs uppercase tracking-wide">
            <tr>
              <th className="text-left px-3 py-2.5 font-semibold">Customer</th>
              <th className="text-left px-3 py-2.5 font-semibold">Tipe</th>
              <th className="text-left px-3 py-2.5 font-semibold">Salesman</th>
              <th className="text-left px-3 py-2.5 font-semibold">Siklus (hari)</th>
              <th className="text-left px-3 py-2.5 font-semibold">Status</th>
              <th className="text-left px-3 py-2.5 font-semibold">Mutasi ke</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {pageRows.map((c) => (
              <tr key={c.id} className="hover:bg-slate-50/60">
                <td className="px-3 py-2 font-medium text-slate-800">{c.nama}
                  <div className="text-[10px] text-slate-400 font-mono">{c.id}</div>
                </td>
                <td className="px-3 py-2"><Badge tone={c.tipe_customer === "Baru" ? "teal" : "slate"}>{c.tipe_customer}</Badge></td>
                <td className="px-3 py-2 text-slate-600">{c.salesmanNama}</td>
                <td className="px-3 py-2">
                  <input
                    type="number" min={1}
                    value={c.order_cycle_days}
                    onChange={(e) => setCycleDays(c.id, e.target.value)}
                    className="w-16 text-xs border border-slate-300 rounded-md px-2 py-1 tabular-nums"
                  />
                </td>
                <td className="px-3 py-2"><Badge tone={c.status_customer === "Aktif" ? "green" : "stone"}>{c.status_customer}</Badge></td>
                <td className="px-3 py-2">
                  <select
                    value={c.salesman_id}
                    onChange={(e) => mutateCustomer(c.id, e.target.value)}
                    className="text-xs rounded-md border border-slate-300 px-2 py-1 bg-white"
                  >
                    {salesmen.map((s) => <option key={s.id} value={s.id}>{s.nama}</option>)}
                  </select>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </Card>

      <div className="flex items-center justify-between mt-3 text-xs text-slate-500">
        <span>Halaman {page + 1} dari {totalPages}</span>
        <div className="flex gap-1.5">
          <Button size="sm" variant="secondary" disabled={page === 0} onClick={() => setPage((p) => p - 1)}>Sebelumnya</Button>
          <Button size="sm" variant="secondary" disabled={page >= totalPages - 1} onClick={() => setPage((p) => p + 1)}>Berikutnya</Button>
        </div>
      </div>
    </div>
  );
}

/* ============================================================
   SALESMAN VIEW  (simulasi tampilan WhatsApp)
   ============================================================ */
function SalesmanView({ salesman, customers, simDate, submitReason, recordNewOrder }) {
  const [openCustomer, setOpenCustomer] = useState(null);
  const [orderSearch, setOrderSearch] = useState("");

  const reminderList = customers
    .filter((c) => c.pendingReminder)
    .sort((a, b) => (b.daysSince ?? 9999) - (a.daysSince ?? 9999));

  const orderCandidates = customers.filter((c) => c.nama.toLowerCase().includes(orderSearch.toLowerCase())).slice(0, 6);

  function exportReminderExcel() {
    const rows = reminderList.map((c, i) => ({
      No: i + 1,
      Customer: c.nama,
      "Hari Belum Order": c.daysSince ?? "-",
      "Siklus Normal (hari)": c.order_cycle_days,
      Status: c.status_customer,
    }));
    const ws = XLSX.utils.json_to_sheet(rows);
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, "Reminder Customer");
    XLSX.writeFile(wb, `Reminder_${salesman?.nama || "Salesman"}_${simDate}.xlsx`);
  }

  function exportReminderText() {
    const lines = [];
    lines.push(`🔔 Reminder Customer Anda [${salesman?.nama || "-"}] — ${fmtDate(simDate)}`);
    lines.push("");
    lines.push(`Ada ${reminderList.length} customer yang sudah melewati siklus order normalnya:`);
    reminderList.forEach((c, i) => {
      const info = c.daysSince === null ? "belum ada order" : `${c.daysSince} hari (siklus normal ${c.order_cycle_days} hari)`;
      lines.push(`${i + 1}. ${c.nama} — ${info}`);
    });
    lines.push("");
    lines.push("Balas dengan alasan: Kalah Harga / Stok Masih Ada / Sudah Bangkrut.");

    const text = lines.join("\n");
    const blob = new Blob([text], { type: "text/plain;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `Reminder_${salesman?.nama || "Salesman"}_${simDate}.txt`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }

  return (
    <div className="grid lg:grid-cols-2 gap-6">
      <div>
        <div className="flex items-center justify-between gap-3 flex-wrap mb-1">
          <SectionHeader title={`Halo, ${salesman?.nama}`} subtitle="Simulasi notifikasi WhatsApp yang kamu terima hari ini." icon={MessageCircle} />
          {reminderList.length > 0 && (
            <div className="flex gap-2">
              <Button size="sm" variant="secondary" onClick={exportReminderText}>
                <FileText className="h-3.5 w-3.5" /> Download Teks (.txt)
              </Button>
              <Button size="sm" variant="secondary" onClick={exportReminderExcel}>
                <Download className="h-3.5 w-3.5" /> Download Excel
              </Button>
            </div>
          )}
        </div>

        <div className="bg-[#e9edef] rounded-2xl p-3 border border-slate-200">
          <div className="bg-white rounded-xl shadow-sm p-4">
            <div className="flex items-center gap-2 mb-2 pb-2 border-b border-slate-100">
              <div className="h-8 w-8 rounded-full bg-teal-700 flex items-center justify-center text-white text-xs font-bold shrink-0">SG</div>
              <div>
                <div className="text-sm font-semibold text-slate-800">SiGula Notif</div>
                <div className="text-[10px] text-slate-400">{fmtDate(simDate)}</div>
              </div>
            </div>

            {reminderList.length === 0 ? (
              <div className="text-sm text-slate-500 py-4 text-center">
                <Sparkles className="h-5 w-5 mx-auto mb-1 text-amber-400" />
                Belum ada notifikasi. Admin belum menekan tombol "Kirim Notifikasi", atau semua customer yang pernah dinotif sudah ditindaklanjuti.
              </div>
            ) : (
              <>
                <p className="text-sm text-slate-700">
                  🔔 Reminder Customer Anda — {reminderList.length} customer sudah melewati siklus order normalnya:
                </p>
                <div className="mt-2.5 space-y-1.5 max-h-[520px] overflow-y-auto pr-1">
                  {reminderList.map((c, i) => (
                    <button
                      key={c.id}
                      onClick={() => setOpenCustomer(c.id)}
                      className="w-full text-left flex items-start justify-between gap-3 text-sm border border-slate-200 rounded-lg px-3.5 py-2.5 hover:bg-teal-50 hover:border-teal-300 transition-colors"
                    >
                      <span className="text-slate-700 leading-snug">{i + 1}. {c.nama}</span>
                      <span className="text-xs text-rose-600 font-semibold tabular-nums whitespace-nowrap shrink-0">
                        {c.daysSince === null ? "belum ada order" : `${c.daysSince}h (siklus ${c.order_cycle_days}h)`}
                      </span>
                    </button>
                  ))}
                </div>
                <p className="text-[11px] text-slate-400 mt-2">Tap salah satu customer untuk isi alasan belum order.</p>
              </>
            )}
          </div>
        </div>

        {openCustomer && (
          <div className="mt-4 bg-[#e9edef] rounded-2xl p-3 border border-slate-200">
            <div className="bg-white rounded-xl shadow-sm p-3.5 max-w-sm">
              {(() => {
                const c = customers.find((x) => x.id === openCustomer);
                return (
                  <>
                    <p className="text-sm text-slate-700">
                      <b>{c.nama}</b> — {c.daysSince === null ? "belum tercatat order" : `${c.daysSince} hari belum order (siklus normal ${c.order_cycle_days} hari)`}.<br />
                      Kenapa belum order lagi?
                    </p>
                    <div className="grid grid-cols-1 gap-1.5 mt-3">
                      {[1, 2, 3].map((k) => (
                        <button
                          key={k}
                          onClick={() => { submitReason(c.id, k); setOpenCustomer(null); }}
                          className={classNames(
                            "text-sm text-center rounded-lg border py-2 font-medium hover:opacity-80 transition-opacity",
                            k === 3 ? "border-rose-300 text-rose-700 bg-rose-50" : "border-teal-300 text-teal-700 bg-teal-50"
                          )}
                        >
                          {REASON_LABELS[k]}
                        </button>
                      ))}
                    </div>
                    <button onClick={() => setOpenCustomer(null)} className="text-xs text-slate-400 mt-2 hover:underline">Batal</button>
                  </>
                );
              })()}
            </div>
          </div>
        )}
      </div>

      <div>
        <SectionHeader title="Catat Order Baru" subtitle="Simulasi ada customer yang order lagi (memicu reaktivasi otomatis jika sebelumnya Inactive)." icon={Send} />
        <Card className="p-4">
          <div className="relative mb-3">
            <Search className="h-3.5 w-3.5 text-slate-400 absolute left-2.5 top-1/2 -translate-y-1/2" />
            <input
              value={orderSearch}
              onChange={(e) => setOrderSearch(e.target.value)}
              placeholder="Cari nama customer..."
              className="w-full text-sm border border-slate-300 rounded-lg pl-8 pr-3 py-2 focus:outline-none focus:ring-2 focus:ring-teal-600/30 focus:border-teal-600"
            />
          </div>
          <div className="space-y-1.5 max-h-80 overflow-auto">

              <div key={c.id} className="flex items-center justify-between text-sm border border-slate-100 rounded-lg px-3 py-2">
                <div>
                  <div className="font-medium text-slate-700">{c.nama}</div>
                  <div className="text-[11px] text-slate-400">
                    Terakhir order: {fmtDate(c.last_order_date)} · <Badge tone={c.status_customer === "Aktif" ? "green" : "stone"}>{c.status_customer}</Badge>
                  </div>
                </div>
                <Button size="sm" variant="primary" onClick={() => recordNewOrder(c.id)}>Catat Order</Button>
              </div>
            ))}
            {orderCandidates.length === 0 && <div className="text-xs text-slate-400 py-3 text-center">Customer tidak ditemukan.</div>}
          </div>
        </Card>

        <Card className="p-4 mt-4">
          <div className="text-sm font-semibold text-slate-700 mb-2">Daftar Customer Saya ({customers.length})</div>
          <div className="max-h-64 overflow-auto space-y-1 text-sm">
            {customers
              .slice()
              .sort((a, b) => (b.daysSince ?? 0) - (a.daysSince ?? 0))
              .map((c) => (
                <div key={c.id} className="flex items-center justify-between border-b border-slate-50 py-1.5">
                  <span className="text-slate-600">{c.nama}</span>
                  <span className="text-xs text-slate-400 tabular-nums">
                    {c.daysSince === null ? "-" : `${c.daysSince}h`} <Badge tone={c.status_customer === "Aktif" ? "green" : "stone"}>{c.status_customer}</Badge>
                  </span>
                </div>
              ))}
          </div>
        </Card>
      </div>
    </div>
  );
}

                  <td className="py-2 font-medium text-slate-800">{s.nama}</td>

                  <td className="py-2 tabular-nums">{flagged > 0 ? <Badge tone="red">{flagged}</Badge> : <span className="text-slate-300">-</span>}</td>
                  <td className="py-2 tabular-nums text-slate-500">{responded}</td>
                </tr>
              );
            })}
          </tbody>
        </table>
      </Card>
    </div>
  );
}
